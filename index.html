import os import yt_dlp def download_media(video_url, folder_name, mode="audio"): current_directory = os.path.dirname(os.path.abspath(__file__)) folder_path = os.path.join(current_directory, folder_name) if not os.path.exists(folder_path): os.makedirs(folder_path) output_path = os.path.join(folder_path, "%(title)s.%(ext)s") if mode == "audio": ydl_opts = { 'format': 'bestaudio/best', 'outtmpl': output_path, 'noplaylist': False, 'postprocessors': [{ 'key': 'FFmpegExtractAudio', 'preferredcodec': 'mp3', 'preferredquality': '192', }], 'quiet': False, } elif mode == "video": ydl_opts = { 'format': 'bestvideo[ext=mp4]+bestaudio[ext=m4a]/mp4', 'outtmpl': output_path, 'noplaylist': False, 'merge_output_format': 'mp4', 'quiet': False, } else: print("Nieznany tryb. Użyj 'audio' lub 'video'.") return try: with yt_dlp.YoutubeDL(ydl_opts) as ydl: ydl.download([video_url]) print("Pobieranie zakończone!") except Exception as e: print(f"Wystąpił błąd: {e}") if __name__ == "__main__": print("Podaj nazwę folderu do zapisu:") folder_name = input("> ").strip() print("Podaj URL filmu lub playlisty:") url = input("> ").strip() print("Co chcesz pobrać? (audio / video)") mode = input("> ").strip().lower() if mode in ["audio", "video"]: download_media(url, folder_name, mode) else: print("Nieprawidłowy wybór. Wybierz 'audio' lub 'video'.")
